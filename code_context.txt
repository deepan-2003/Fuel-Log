"""Pure-pandas helpers: derived columns, monthly roll-ups, period handling."""
from __future__ import annotations

import calendar
from datetime import date, timedelta

import pandas as pd

METRICS = {
    "Fuel cost (₹)": ("amount", "money"),
    "Total KM": ("total_km", "int"),
    "Fuel filled (L)": ("fuel_litres", "dec"),
    "No. of fills": ("fills", "int"),
    "Avg KM per litre": ("mileage", "dec"),
    "Cost per KM (₹)": ("cost_per_km", "dec"),
}

PERIOD_CHOICES = [
    "All time",
    "This month",
    "Last month",
    "Last 30 days",
    "This financial year (Apr-Mar)",
    "Specific month",
    "Custom range",
]


# --------------------------------------------------------------------------- #
def _ratio(num: pd.Series, den: pd.Series) -> pd.Series:
    """num / den, NaN wherever the denominator is zero or missing."""
    return num / den.where(den > 0)


def enrich(df: pd.DataFrame) -> pd.DataFrame:
    """Add total_km, mileage, cost columns and month keys to an entries frame."""
    df = df.copy()
    df["entry_date"] = pd.to_datetime(df["entry_date"])
    df["bill_date"] = pd.to_datetime(df["bill_date"], errors="coerce")
    df["total_km"] = df["closing_km"] - df["starting_km"]
    df["mileage"] = _ratio(df["total_km"], df["fuel_litres"])
    df["rate_per_litre"] = _ratio(df["amount"], df["fuel_litres"])
    df["cost_per_km"] = _ratio(df["amount"], df["total_km"])
    df["month_key"] = df["entry_date"].dt.strftime("%Y-%m")
    df["month_label"] = df["entry_date"].dt.strftime("%b %Y")
    return df


def month_label(key: str) -> str:
    return pd.Timestamp(key + "-01").strftime("%b %Y")


def totals(df: pd.DataFrame) -> dict:
    km = float(df["total_km"].sum()) if len(df) else 0.0
    fuel = float(df["fuel_litres"].sum()) if len(df) else 0.0
    amt = float(df["amount"].sum()) if len(df) else 0.0
    return {
        "fills": int(len(df)),
        "total_km": km,
        "fuel_litres": fuel,
        "amount": amt,
        "mileage": km / fuel if fuel > 0 else None,
        "cost_per_km": amt / km if km > 0 else None,
        "rate_per_litre": amt / fuel if fuel > 0 else None,
    }


def monthly_summary(df: pd.DataFrame, by_vehicle: bool = True) -> pd.DataFrame:
    cols = ["reg_no", "month_key", "month_label", "fills", "total_km",
            "fuel_litres", "amount", "mileage", "cost_per_km", "rate_per_litre"]
    if df.empty:
        return pd.DataFrame(columns=cols)
    keys = (["reg_no"] if by_vehicle else []) + ["month_key", "month_label"]
    g = df.groupby(keys, as_index=False).agg(
        fills=("id", "count"),
        total_km=("total_km", "sum"),
        fuel_litres=("fuel_litres", "sum"),
        amount=("amount", "sum"),
    )
    g["mileage"] = _ratio(g["total_km"], g["fuel_litres"])
    g["cost_per_km"] = _ratio(g["amount"], g["total_km"])
    g["rate_per_litre"] = _ratio(g["amount"], g["fuel_litres"])
    if not by_vehicle:
        g.insert(0, "reg_no", "All vehicles")
    return g.sort_values(["reg_no", "month_key"]).reset_index(drop=True)


def vehicle_summary(df: pd.DataFrame) -> pd.DataFrame:
    if df.empty:
        return pd.DataFrame(columns=["reg_no", "fills", "total_km", "fuel_litres",
                                     "amount", "mileage", "cost_per_km", "last_fill"])
    g = df.groupby("reg_no", as_index=False).agg(
        fills=("id", "count"),
        total_km=("total_km", "sum"),
        fuel_litres=("fuel_litres", "sum"),
        amount=("amount", "sum"),
        last_fill=("entry_date", "max"),
    )
    g["mileage"] = _ratio(g["total_km"], g["fuel_litres"])
    g["cost_per_km"] = _ratio(g["amount"], g["total_km"])
    return g.sort_values("reg_no").reset_index(drop=True)


def month_keys_between(df: pd.DataFrame) -> list[str]:
    """Continuous list of YYYY-MM keys from the first to the last entry."""
    if df.empty:
        return []
    lo, hi = df["entry_date"].min(), df["entry_date"].max()
    return [str(p) for p in pd.period_range(lo, hi, freq="M")]


def _sum_matrix(d: pd.DataFrame, col: str, keys: list[str]) -> pd.DataFrame:
    p = (d.pivot_table(index="reg_no", columns="month_key", values=col,
                       aggfunc="sum", fill_value=0)
          .reindex(columns=keys, fill_value=0))
    p["Total"] = p.sum(axis=1)
    p.loc["All vehicles"] = p.sum(axis=0)
    return p


def month_matrix(df: pd.DataFrame, metric: str) -> pd.DataFrame:
    """Vehicles (+ 'All vehicles') as rows, months (+ 'Total') as columns."""
    if df.empty:
        return pd.DataFrame()
    d = df.copy()
    d["fills"] = 1
    keys = month_keys_between(d)
    field = METRICS[metric][0]
    if field in ("fills", "total_km", "fuel_litres", "amount"):
        return _sum_matrix(d, field, keys)
    km = _sum_matrix(d, "total_km", keys)
    if field == "mileage":
        return km / _sum_matrix(d, "fuel_litres", keys).where(lambda x: x > 0)
    return _sum_matrix(d, "amount", keys) / km.where(km > 0)


# --------------------------------------------------------------------------- #
def resolve_period(choice: str, today: date | None = None, custom=(None, None),
                   month_key: str | None = None):
    """Translate a period choice into an inclusive (start, end) date pair."""
    today = today or date.today()

    def month_bounds(y: int, m: int):
        return date(y, m, 1), date(y, m, calendar.monthrange(y, m)[1])

    if choice == "This month":
        return month_bounds(today.year, today.month)
    if choice == "Last month":
        first = today.replace(day=1) - timedelta(days=1)
        return month_bounds(first.year, first.month)
    if choice == "Last 30 days":
        return today - timedelta(days=29), today
    if choice.startswith("This financial year"):
        y = today.year if today.month >= 4 else today.year - 1
        return date(y, 4, 1), date(y + 1, 3, 31)
    if choice == "Specific month" and month_key:
        y, m = (int(x) for x in month_key.split("-"))
        return month_bounds(y, m)
    if choice == "Custom range":
        return custom
    return None, None

def _norm(v):
    """Normalise a cell so edited and original values compare cleanly."""
    if v is None or (not isinstance(v, str) and pd.isna(v)):
        return None
    if isinstance(v, (pd.Timestamp, datetime)):
        return v.date()
    if isinstance(v, str):
        return v.strip()
    if isinstance(v, float):
        return round(v, 4)
    return v

def _save_entry(vid: int, start_key: str):
    s = st.session_state
    try:
        bill_date = s["ne_date"] if s.get("ne_bill_same", True) else s["ne_bill_date"]
        db.add_entry(vid, s["ne_date"], s["ne_close"], s[start_key], s["ne_fuel"],

        s["flash"] = ("error", str(exc))
        return
    s["flash"] = ("success", "Entry saved ✔")
    _reset_entry_form()

"""SQLite persistence for vehicles and fuel entries."""
from __future__ import annotations

import os
import sqlite3
from contextlib import contextmanager
from datetime import date, datetime, timedelta
from pathlib import Path

import pandas as pd

import analytics

DB_PATH = Path(os.environ.get("FUEL_LOG_DB", Path(__file__).with_name("fuel_log.db")))

SCHEMA = """
CREATE TABLE IF NOT EXISTS vehicles(
    id            INTEGER PRIMARY KEY AUTOINCREMENT,
    reg_no        TEXT NOT NULL UNIQUE,
    name          TEXT DEFAULT '',
    driver        TEXT DEFAULT '',
    tank_capacity REAL DEFAULT 0,
    fc_date       TEXT,
    active        INTEGER NOT NULL DEFAULT 1
);
CREATE TABLE IF NOT EXISTS fuel_entries(
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    vehicle_id  INTEGER NOT NULL REFERENCES vehicles(id) ON DELETE CASCADE,
    entry_date  TEXT NOT NULL,
    closing_km  REAL NOT NULL,
    starting_km REAL NOT NULL,
    fuel_litres REAL NOT NULL,
    bill_no     TEXT DEFAULT '',
    bill_date   TEXT,
    amount      REAL NOT NULL DEFAULT 0,
    remarks     TEXT DEFAULT '',
    created_at  TEXT DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX IF NOT EXISTS ix_entries_vehicle_date ON fuel_entries(vehicle_id, entry_date);
CREATE TABLE IF NOT EXISTS settings(key TEXT PRIMARY KEY, value TEXT);
"""


@contextmanager
def connect():
    con = sqlite3.connect(DB_PATH)
    con.execute("PRAGMA foreign_keys = ON")
    try:
        yield con
        con.commit()
    finally:
        con.close()


def init_db() -> None:
    with connect() as con:
        con.executescript(SCHEMA)


# ------------------------------------------------------------------ helpers --
def _iso(d):
    """Any date-like value -> 'YYYY-MM-DD' (or None when empty)."""
    if d is None:
        return None
    if isinstance(d, str):
        d = d.strip()
        if not d:
            return None
        return pd.to_datetime(d, dayfirst=True).date().isoformat()
    if pd.isna(d):
        return None
    if isinstance(d, (datetime, pd.Timestamp)):
        return d.date().isoformat()
    if isinstance(d, date):
        return d.isoformat()
    raise ValueError(f"Unrecognised date: {d!r}")


def validate_entry(closing, starting, fuel, amount) -> None:
    if closing is None or starting is None or pd.isna(closing) or pd.isna(starting):
        raise ValueError("Closing and starting KM are required.")
    if starting < 0:
        raise ValueError("Starting KM cannot be negative.")
    if closing <= starting:
        raise ValueError("Closing KM must be greater than Starting KM.")
    if fuel is None or pd.isna(fuel) or fuel <= 0:
        raise ValueError("Fuel filled must be greater than 0 litres.")
    if amount is None or pd.isna(amount) or amount < 0:
        raise ValueError("Amount cannot be negative.")

    with connect() as con:

    return row[0] if row else default

    with connect() as con:
        con.execute("INSERT INTO settings(key,value) VALUES(?,?) "
                    "ON CONFLICT(key) DO UPDATE SET value=excluded.value", (key, value))

    df["active"] = df["active"].astype(bool)
    return df


def _tank(v) -> float:
    return 0.0 if v is None or pd.isna(v) else float(v)


def add_vehicle(reg_no: str, name: str = "", driver: str = "",
                tank_capacity: float = 0, fc_date=None) -> int:
    reg_no = (reg_no or "").strip().upper()
    if not reg_no:
        raise ValueError("Vehicle registration number is required.")
    try:
        with connect() as con:
            cur = con.execute(
                "INSERT INTO vehicles(reg_no,name,driver,tank_capacity,fc_date) VALUES(?,?,?,?,?)",
                (reg_no, (name or "").strip(), (driver or "").strip(),
                 _tank(tank_capacity), _iso(fc_date)))
            return cur.lastrowid
    except sqlite3.IntegrityError:
        raise ValueError(f"Vehicle {reg_no} already exists.") from None

                   fc_date, active: bool) -> None:

             _iso(fc_date), 1 if active else 0, int(vehicle_id)))

def update_entry(entry_id, entry_date, closing_km, starting_km, fuel_litres,

def entries_df(vehicle_ids=None, start=None, end=None) -> pd.DataFrame:
    q = """SELECT e.id, e.vehicle_id, v.reg_no, e.entry_date, e.closing_km, e.starting_km,
                  e.fuel_litres, e.bill_no, e.bill_date, e.amount, e.remarks
           FROM fuel_entries e JOIN vehicles v ON v.id = e.vehicle_id WHERE 1=1"""
    params: list = []
    if vehicle_ids:
        q += f" AND e.vehicle_id IN ({','.join('?' * len(vehicle_ids))})"

    with connect() as con:

def _norm_header(h) -> str:
    return "".join(ch for ch in str(h).lower() if ch.isalnum())


def standardise_import(raw: pd.DataFrame) -> pd.DataFrame:
    """Rename the uploaded sheet's headers to IMPORT_COLUMNS; raise if required ones are missing."""
    lookup = {a: std for std, names in _ALIASES.items() for a in names}
    rename = {}
    for col in raw.columns:
        std = lookup.get(_norm_header(col))
        if std and std not in rename.values():
            rename[col] = std
    df = raw.rename(columns=rename)
    missing = [c for c in ["Vehicle", "Date", "Closing KM", "Starting KM", "Fuel (L)", "Amount"]
               if c not in df.columns]
    if missing:
        raise ValueError("Missing column(s): " + ", ".join(missing))
    for c in IMPORT_COLUMNS:
        if c not in df.columns:
            df[c] = None
    return df[IMPORT_COLUMNS].dropna(how="all")


def ensure_vehicles(infos) -> int:
    """Create any vehicle (with driver, tank, FC date) that does not exist yet."""
    made = 0
    for v in infos:
        if vehicle_id_by_reg(v["reg_no"]) is None:
            add_vehicle(v["reg_no"], v.get("name", ""), v.get("driver", ""),
                        v.get("tank_capacity", 0), v.get("fc_date"))
            made += 1
    return made


def import_entries(df: pd.DataFrame, create_missing: bool = True):
    """Insert rows from a standardised frame. Returns (inserted, skipped, messages)."""
    inserted, skipped, msgs = 0, 0, []
    for i, row in df.iterrows():
        line = i + 2  # header is line 1 in the source sheet
        try:
            reg = str(row["Vehicle"]).strip().upper()
            vid = vehicle_id_by_reg(reg)
            if vid is None:
                if not create_missing or not reg or reg == "NAN":
                    raise ValueError(f"unknown vehicle '{reg}'")
                vid = add_vehicle(reg)
            bill = "" if pd.isna(row["Bill No"]) else str(row["Bill No"]).strip()
            if bill.endswith(".0"):
                bill = bill[:-2]
            remarks = "" if pd.isna(row["Remarks"]) else str(row["Remarks"])
            if _is_duplicate(vid, row):
                skipped += 1
                msgs.append(f"Line {line}: skipped (same vehicle, date and closing KM already exist)")
                continue
            add_entry(vid, row["Date"], float(row["Closing KM"]), float(row["Starting KM"]),
                      float(row["Fuel (L)"]), bill, row["Bill Date"],
                      float(row["Amount"]) if not pd.isna(row["Amount"]) else 0.0, remarks)
            inserted += 1
        except Exception as exc:  # keep importing the remaining rows
            skipped += 1
            msgs.append(f"Line {line}: {exc}")
    return inserted, skipped, msgs


def _is_duplicate(vehicle_id, row) -> bool:
    with connect() as con:
        return con.execute(
            "SELECT 1 FROM fuel_entries WHERE vehicle_id=? AND entry_date=? AND closing_km=?",
            (int(vehicle_id), _iso(row["Date"]), float(row["Closing KM"]))).fetchone() is not None

    c.font = Font(name=FONT, size=size, bold=bold, italic=italic, color=color)
    c.alignment = Alignment(horizontal=h, vertical=v, wrap_text=wrap)
    if fill:
        c.fill = _fill(fill)
    if fmt:
        c.number_format = fmt
    if border:
        c.border = border
    return c


def _merge(ws, row, c1, c2, value=None, **kw):
    """Merge a horizontal range, styling every cell so borders/fills render fully."""
    for c in range(c1, c2 + 1):
        _put(ws, row, c, value if c == c1 else None, **kw)
    if c2 > c1:
        ws.merge_cells(start_row=row, start_column=c1, end_row=row, end_column=c2)


def _banner(ws, ncol, org, subtitle, meta):
    _merge(ws, 1, 1, ncol, org, bold=True, size=16, color="FFFFFF", fill=NAVY, border=None)
    ws.row_dimensions[1].height = 30
    _merge(ws, 2, 1, ncol, subtitle, bold=True, size=12, color="FFFFFF", fill=BLUE, border=None)
    ws.row_dimensions[2].height = 22
    _merge(ws, 3, 1, ncol, meta, italic=True, size=9, color="595959", border=None, h="left")


def _safe_sheet_name(name: str, used: set[str]) -> str:
    base = re.sub(r"[\[\]\:\*\?\/\\']", "-", name).strip()[:31] or "Vehicle"
    cand, n = base, 2
    while cand.lower() in used or cand.lower() == "summary":
        suffix = f" ({n})"
        cand = base[:31 - len(suffix)] + suffix
        n += 1
    used.add(cand.lower())
    return cand


def _page_setup(ws, title_rows=None):

    ws.print_options.horizontalCentered = True
    ws.page_margins.left = ws.page_margins.right = 0.4
    ws.oddFooter.center.text = "Page &P of &N"
    ws.oddFooter.center.size = 8
    if title_rows:
        ws.print_title_rows = title_rows

VEH_HEADERS = ["Date", "Closing KM", "Starting KM", "Total KM", "Fuel Filled (L)",

def _vehicle_sheet(wb, title, veh, g, org, period, generated):
    ws = wb.create_sheet(title)
    ws.sheet_view.showGridLines = False
    ws.sheet_properties.tabColor = BLUE
    n = len(VEH_HEADERS)
    for i, w in enumerate(VEH_WIDTHS, 1):
        ws.column_dimensions[L(i)].width = w

    _banner(ws, n, org, f"Fuel Consumption Log  -  {veh['reg_no']}",
            f"Period: {period}     |     Generated: {generated:%d-%m-%Y %H:%M}")

    tank = float(veh.get("tank_capacity") or 0)
    fc = veh.get("fc_date")
    info = [
        (4, "Vehicle No.", veh["reg_no"], "Driver", veh.get("driver") or "-"),
        (5, "Description", veh.get("name") or "-", "Tank Capacity",
         f"{tank:g} L" if tank else "-"),
        (6, "FC Due", fc.strftime("%d-%m-%Y") if pd.notna(fc) else "-", "Period", period),
    ]
    for r, l1, v1, l2, v2 in info:
        _put(ws, r, 1, l1, bold=True, fill=LIGHT, h="left")
        _merge(ws, r, 2, 4, v1, h="left")
        _put(ws, r, 5, l2, bold=True, fill=LIGHT, h="left", wrap=True)
        _merge(ws, r, 6, n, v2, h="left")
    ws.row_dimensions[8].height = 34
    for i, h in enumerate(VEH_HEADERS, 1):
        _put(ws, 8, i, h, bold=True, color="FFFFFF", fill=HEAD, wrap=True)
    ws.freeze_panes = "A9"

    r = 9
    sub_rows: dict[str, int] = {}
    for mk, grp in g.groupby("month_key", sort=True):
        label = grp["month_label"].iloc[0]
        _merge(ws, r, 1, n, label, bold=True, color=NAVY, fill=BAND, h="left", size=11)
        ws.row_dimensions[r].height = 20
        r += 1
        first = r
        for _, e in grp.sort_values(["entry_date", "id"]).iterrows():
            bill_date = e["bill_date"].date() if pd.notna(e["bill_date"]) else None
            _put(ws, r, 1, e["entry_date"].date(), fmt=F_DATE)
            _put(ws, r, 2, float(e["closing_km"]), fmt="#,##0")
            _put(ws, r, 3, float(e["starting_km"]), fmt="#,##0")
            _put(ws, r, 4, f"=B{r}-C{r}", fmt="#,##0")
            _put(ws, r, 5, float(e["fuel_litres"]), fmt="0.00")
            _put(ws, r, 6, f"=IFERROR(D{r}/E{r},0)", fmt="0.00")
            _put(ws, r, 7, e["bill_no"] or None)
            _put(ws, r, 8, bill_date, fmt=F_DATE)
            _put(ws, r, 9, float(e["amount"]), fmt=F_MONEY, h="right")
            _put(ws, r, 10, f"=IFERROR(I{r}/D{r},0)", fmt=F_MONEY, h="right")
            _put(ws, r, 11, e["remarks"] or None, h="left", wrap=True)
            r += 1
        last = r - 1
        _merge(ws, r, 1, 3, f"Sub Total  -  {label}", bold=True, color=NAVY, fill=SUB,
               h="left", border=TOP_RULE)
        _put(ws, r, 4, f"=SUM(D{first}:D{last})", bold=True, fill=SUB, fmt="#,##0", border=TOP_RULE)
        _put(ws, r, 5, f"=SUM(E{first}:E{last})", bold=True, fill=SUB, fmt="0.00", border=TOP_RULE)
        _put(ws, r, 6, f"=IFERROR(D{r}/E{r},0)", bold=True, fill=SUB, fmt="0.00", border=TOP_RULE)
        _put(ws, r, 7, f"=COUNT(D{first}:D{last})", bold=True, fill=SUB, fmt='0" fills"', border=TOP_RULE)
        _put(ws, r, 8, None, fill=SUB, border=TOP_RULE)
        _put(ws, r, 9, f"=SUM(I{first}:I{last})", bold=True, fill=SUB, fmt=F_MONEY, h="right", border=TOP_RULE)
        _put(ws, r, 10, f"=IFERROR(I{r}/D{r},0)", bold=True, fill=SUB, fmt=F_MONEY, h="right", border=TOP_RULE)
        _put(ws, r, 11, None, fill=SUB, border=TOP_RULE)
        sub_rows[mk] = r
        r += 1
        ws.row_dimensions[r].height = 8
        r += 1

    refs = list(sub_rows.values())

    def tot(col):
        return "=SUM(" + ",".join(f"{col}{x}" for x in refs) + ")"

    _merge(ws, r, 1, 3, "GRAND TOTAL", bold=True, color="FFFFFF", fill=NAVY, h="left", size=11)
    ws.row_dimensions[r].height = 22
    _put(ws, r, 4, tot("D"), bold=True, color="FFFFFF", fill=NAVY, fmt="#,##0")
    _put(ws, r, 5, tot("E"), bold=True, color="FFFFFF", fill=NAVY, fmt="0.00")
    _put(ws, r, 6, f"=IFERROR(D{r}/E{r},0)", bold=True, color="FFFFFF", fill=NAVY, fmt="0.00")
    _put(ws, r, 7, tot("G"), bold=True, color="FFFFFF", fill=NAVY, fmt='0" fills"')
    _put(ws, r, 8, None, fill=NAVY)
    _put(ws, r, 9, tot("I"), bold=True, color="FFFFFF", fill=NAVY, fmt=F_MONEY, h="right")
    _put(ws, r, 10, f"=IFERROR(I{r}/D{r},0)", bold=True, color="FFFFFF", fill=NAVY, fmt=F_MONEY, h="right")
    _put(ws, r, 11, None, fill=NAVY)
    r += 2
    _merge(ws, r, 1, n, "Avg KM/L = Total KM / Fuel filled.   Cost / KM = Amount / Total KM.   "

           italic=True, size=8, color="7F7F7F", border=None, h="left")

def _summary_sheet(ws, vehicles, sub_rows, months, org, period, generated):
    ws.sheet_view.showGridLines = False
    ws.sheet_properties.tabColor = NAVY
    nm = len(months)
    ncol = 3 + nm
    ws.column_dimensions["A"].width = 24
    ws.column_dimensions["B"].width = 18
    for c in range(3, ncol):
        ws.column_dimensions[L(c)].width = 14
    ws.column_dimensions[L(ncol)].width = 17

    _banner(ws, ncol, org, "Fuel Consumption Summary  -  Month-wise by Vehicle",
            f"Period: {period}     |     Generated: {generated:%d-%m-%Y %H:%M}")
    hdr = 5
    ws.row_dimensions[hdr].height = 26
    _put(ws, hdr, 1, "Vehicle", bold=True, color="FFFFFF", fill=HEAD)
    _put(ws, hdr, 2, "Metric", bold=True, color="FFFFFF", fill=HEAD)
    for i, mk in enumerate(months):
        _put(ws, hdr, 3 + i, an.month_label(mk), bold=True, color="FFFFFF", fill=HEAD)
    _put(ws, hdr, ncol, "Total", bold=True, color="FFFFFF", fill=NAVY)
    ws.freeze_panes = ws.cell(hdr + 1, 3)

    metrics = [("No. of fills", F_INT, "G"), ("Total KM", F_KM, "D"), ("Fuel (L)", F_DEC, "E"),
               (f"Amount ({CUR})", F_MONEY, "I"), ("Avg KM/L", F_DEC, None), (f"Cost / KM ({CUR})", F_MONEY, None)]
    first_m, last_m = L(3), L(2 + nm)

    def write_block(r0, label, sources, fill_a, strong=False):
        """sources[(row_offset, col_idx)] -> formula body; ratio rows derive from same column."""
        fill_row = SUB if strong else None
        merge_end = r0 + len(metrics) - 1
        _put(ws, r0, 1, label, bold=True, color="FFFFFF" if strong else NAVY, fill=fill_a, wrap=True)
        for k in range(1, len(metrics)):
            _put(ws, r0 + k, 1, None, fill=fill_a)
        ws.merge_cells(start_row=r0, start_column=1, end_row=merge_end, end_column=1)
        for k, (name, fmt, _src) in enumerate(metrics):
            r = r0 + k
            _put(ws, r, 2, name, bold=strong, h="left", fill=fill_row, italic=k >= 4)
            for ci in range(3, ncol + 1):
                col = L(ci)
                if k < 4:
                    val = (f"=SUM({first_m}{r}:{last_m}{r})" if ci == ncol else sources(k, ci))
                elif k == 4:
                    val = f"=IFERROR({col}{r0 + 1}/{col}{r0 + 2},0)"
                else:
                    val = f"=IFERROR({col}{r0 + 3}/{col}{r0 + 1},0)"
                _put(ws, r, ci, val, fmt=fmt, bold=strong or ci == ncol, fill=fill_row or (BAND if ci == ncol else None),
                     h="right", italic=k >= 4)
        return r0

    r = hdr + 1
    blocks = []
    for veh in vehicles:
        rows = sub_rows[veh["reg_no"]]
        sheet = veh["sheet"]

        def src(k, ci, rows=rows, sheet=sheet):
            mk = months[ci - 3]
            if mk not in rows:
                return 0
            return f"='{sheet}'!{metrics[k][2]}{rows[mk]}"

        write_block(r, veh["reg_no"], src, BAND)
        blocks.append((veh["reg_no"], r))
        r += len(metrics)
        ws.row_dimensions[r].height = 6
        r += 1

    if len(blocks) > 1:
        def fleet(k, ci):
            col = L(ci)
            return "=SUM(" + ",".join(f"{col}{b + k}" for _, b in blocks) + ")"
        write_block(r, "ALL VEHICLES", fleet, NAVY, strong=True)
        r += len(metrics) + 1

    _merge(ws, r, 1, ncol, "Figures link to the sub totals on each vehicle sheet. "
                           "Avg KM/L = Total KM / Fuel; Cost / KM = Amount / Total KM.",
           italic=True, size=8, color="7F7F7F", border=None, h="left")
    r += 2

    # Monthly fuel-cost chart
    ch = BarChart()
    ch.type = "col"
    ch.grouping = "clustered"
    ch.title = f"Monthly Fuel Cost ({CUR}) by Vehicle"
    ch.roundedCorners = False
    ch.title.overlay = False
    ch.style = 10
    for reg, b in blocks:
        s = Series(Reference(ws, min_col=3, max_col=2 + nm, min_row=b + 3, max_row=b + 3), title=reg)
        ch.series.append(s)
    ch.set_categories(Reference(ws, min_col=3, max_col=2 + nm, min_row=hdr, max_row=hdr))
    ch.y_axis.title = f"Amount ({CUR})"
    ch.y_axis.title.overlay = False
    ch.y_axis.numFmt = "#,##0"
    ch.y_axis.majorGridlines = ch.y_axis.majorGridlines
    ch.x_axis.delete = False
    ch.y_axis.delete = False
    ch.legend.position = "b"
    ch.legend.overlay = False
    ch.height, ch.width = 9, max(18, min(34, 3 + nm * 3.2))
    ws.add_chart(ch, f"A{r}")
    _page_setup(ws)

def build_workbook(entries: pd.DataFrame, vehicles: pd.DataFrame, org_name: str,
                   period_label: str) -> bytes:
    """entries must come from db.entries_df(); vehicles from db.vehicles_df()."""
    generated = datetime.now()
    wb = Workbook()
    summary = wb.active
    summary.title = "Summary"

    vmap = vehicles.set_index("reg_no").to_dict("index")
    used: set[str] = set()
    veh_list, sub_rows = [], {}
    for reg in sorted(entries["reg_no"].unique()):
        g = entries[entries["reg_no"] == reg]
        info = {"reg_no": reg, **vmap.get(reg, {})}
        info["sheet"] = _safe_sheet_name(reg, used)
        sub_rows[reg] = _vehicle_sheet(wb, info["sheet"], info, g, org_name, period_label, generated)
        veh_list.append(info)

    _summary_sheet(summary, veh_list, sub_rows, an.month_keys_between(entries),
                   org_name, period_label, generated)

    wb.properties.title = "Fuel Consumption Report"
    wb.properties.creator = org_name
    buf = io.BytesIO()
    wb.save(buf)
    return buf.getvalue()

IMPORT_COLS = ["Vehicle", "Date", "Closing KM", "Starting KM", "Fuel (L)",
               "Bill No", "Bill Date", "Amount", "Remarks"]


def _clean(v):
    return "" if v is None or str(v).strip() in ("-", "None") else str(v).strip()


def _is_num(v) -> bool:
    return isinstance(v, (int, float)) and not isinstance(v, bool)


def parse_report(file):
    """Read a workbook produced by build_workbook().

    Returns (entries_frame, vehicle_infos). The frame uses the import column names;
    sub total / grand total / month-band rows are ignored. Returns (empty, []) when the
    file is not in the report layout.
    """
    wb = load_workbook(file, data_only=True)
    rows, infos = [], []
    for ws in wb.worksheets:
        if ws.title == "Summary" or str(ws["A8"].value or "").strip() != "Date":
            continue
        reg = _clean(ws["B4"].value).upper()
        if not reg:
            continue
        tank = re.search(r"[\d.]+", _clean(ws["F5"].value))
        infos.append({"reg_no": reg, "name": _clean(ws["B5"].value), "driver": _clean(ws["F4"].value),
                      "tank_capacity": float(tank.group()) if tank else 0.0,
                      "fc_date": _clean(ws["B6"].value) or None})
        for r in ws.iter_rows(min_row=9, max_col=11, values_only=True):
            d = r[0]
            if isinstance(d, (datetime, date)) and _is_num(r[1]) and _is_num(r[2]) and _is_num(r[4]):
                bill = r[6]
                if isinstance(bill, float) and bill.is_integer():
                    bill = int(bill)
                rows.append([reg, d, r[1], r[2], r[4], _clean(bill), r[7], r[8] if _is_num(r[8]) else 0, _clean(r[10])])
    return pd.DataFrame(rows, columns=IMPORT_COLS), infos

import os

# Folder to scan
SOURCE_FOLDER = r"C:\Users\ELCOT\Downloads\files"

# Output text file
OUTPUT_FILE = "code_context.txt"

# File extensions to include
CODE_EXTENSIONS = {
    ".py", ".js", ".jsx", ".ts", ".tsx",
    ".html", ".css", ".scss",
    ".java", ".cpp", ".c", ".h", ".cs",
    ".php", ".rb", ".go", ".rs",
    ".sql", ".sh", ".bat",
    ".json", ".xml", ".yaml", ".yml",
    ".md", ".txt", ".ipynb",
    ".dart", ".swift", ".kt",
    ".vue", ".svelte"
}

# Folders to ignore
IGNORE_DIRS = {
    ".git", "__pycache__", "node_modules",
    ".venv", "venv", "dist", "build",
    ".idea", ".vscode"
}

def export_code_context():
    with open(OUTPUT_FILE, "w", encoding="utf-8") as output:

        for root, dirs, files in os.walk(SOURCE_FOLDER):

            # Skip unwanted folders
            dirs[:] = [
                d for d in dirs if d not in IGNORE_DIRS
            ]

            for filename in sorted(files):
                ext = os.path.splitext(filename)[1].lower()

                if ext not in CODE_EXTENSIONS:
                    continue

                file_path = os.path.join(root, filename)
                relative_path = os.path.relpath(
                    file_path, SOURCE_FOLDER
                )

                try:
                    with open(
                        file_path, "r", encoding="utf-8"
                    ) as file:
                        content = file.read()

                    output.write("\n")
                    output.write("=" * 80 + "\n")
                    output.write(f"FILE: {relative_path}\n")
                    output.write("=" * 80 + "\n\n")
                    output.write(content)
                    output.write("\n\n")

                    print(f"Exported: {relative_path}")

                except Exception as e:
                    print(f"Skipped: {relative_path} - {e}")

    print(f"\nExport completed: {OUTPUT_FILE}")


if __name__ == "__main__":
    export_code_context()
